cmake_minimum_required(VERSION 3.18)
project(ni26_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED TORCH_ROOT)
  message(FATAL_ERROR "Set TORCH_ROOT to the existing compatible LibTorch/PyTorch installation")
endif()
if(NOT DEFINED TORCH_CXX11_ABI)
  message(FATAL_ERROR "Set TORCH_CXX11_ABI from the target worker build; never guess its ABI")
endif()
add_library(ni26_base SHARED src/artifact.cpp src/math.cpp src/filter.cpp src/core.cpp
  src/tail.cpp src/tail_em.cpp src/tail_tsem.cpp)
target_include_directories(ni26_base PUBLIC include)
target_include_directories(ni26_base SYSTEM PUBLIC "${TORCH_ROOT}/include"
  "${TORCH_ROOT}/include/torch/csrc/api/include")
target_compile_definitions(ni26_base PUBLIC _GLIBCXX_USE_CXX11_ABI=${TORCH_CXX11_ABI})
target_compile_options(ni26_base PRIVATE -O2 -Wall -Wextra)
target_link_directories(ni26_base PUBLIC "${TORCH_ROOT}/lib")
target_link_libraries(ni26_base PUBLIC torch torch_cpu c10)
set_target_properties(ni26_base PROPERTIES BUILD_RPATH "${TORCH_ROOT}/lib" INSTALL_RPATH "$ORIGIN")
if(EXISTS "${TORCH_ROOT}/lib/libtorch_cuda.so")
  target_link_libraries(ni26_base PUBLIC torch_cuda c10_cuda)
endif()
add_executable(ni26_probe tests/probe.cpp)
target_link_libraries(ni26_probe PRIVATE ni26_base)
target_sources(ni26_base PRIVATE src/gates.cpp src/personal.cpp src/routing.cpp
  src/count_filter.cpp src/causal_personal.cpp)
add_executable(ni26_routing_source tests/routing_source.cpp)
target_link_libraries(ni26_routing_source PRIVATE ni26_base)
include(cmake/linear.cmake)
include(cmake/core_training.cmake)
add_library(ni26_runtime SHARED src/runtime.cpp src/api.cpp src/networks.cpp
  src/networks_input.cpp src/networks_models.cpp src/networks_extras.cpp
  src/finetune.cpp src/finetune_random.cpp src/session_bn.cpp
  src/training_fbc.cpp src/training.cpp src/hash.cpp)
target_link_libraries(ni26_runtime PUBLIC ni26_base ni26_initialization)
target_compile_options(ni26_runtime PRIVATE -O2 -Wall -Wextra)
set_target_properties(ni26_runtime PROPERTIES BUILD_RPATH "${TORCH_ROOT}/lib" INSTALL_RPATH "$ORIGIN")
if(EXISTS "${TORCH_ROOT}/lib/libtorch_cuda.so")
  if(NOT DEFINED CUDA_INCLUDE_DIR OR NOT EXISTS "${CUDA_INCLUDE_DIR}/cuda_runtime_api.h")
    message(FATAL_ERROR "Set CUDA_INCLUDE_DIR to the matching existing CUDA headers")
  endif()
  target_include_directories(ni26_runtime SYSTEM PRIVATE "${CUDA_INCLUDE_DIR}")
  target_compile_definitions(ni26_runtime PRIVATE NI26_WITH_CUDA)
endif()
add_executable(ni26_neural_source tests/neural_source.cpp)
target_link_libraries(ni26_neural_source PRIVATE ni26_runtime)
add_executable(ni26_runtime_source tests/runtime_source.cpp)
target_link_libraries(ni26_runtime_source PRIVATE ni26_runtime)
add_executable(ni26_train_raw tools/train_raw.cpp)
target_link_libraries(ni26_train_raw PRIVATE ni26_runtime)
add_executable(ni26_neural_training_source tests/neural_training_source.cpp)
target_link_libraries(ni26_neural_training_source PRIVATE ni26_runtime)
add_executable(ni26_hash_test tests/hash.cpp src/hash.cpp)
target_include_directories(ni26_hash_test PRIVATE include)
add_library(ni26_initialization STATIC src/initialization.cpp)
target_link_libraries(ni26_initialization PUBLIC ni26_base)
target_compile_options(ni26_initialization PRIVATE -O2 -Wall -Wextra)
add_executable(ni26_init_raw tools/init_raw.cpp)
target_link_libraries(ni26_init_raw PRIVATE ni26_initialization)
add_executable(ni26_initialization_source tests/initialization_source.cpp)
target_link_libraries(ni26_initialization_source PRIVATE ni26_initialization)
